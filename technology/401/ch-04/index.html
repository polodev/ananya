Pointer = ঠিকানার কাগজ

একটা variable-এর মান নয়, তার মেমোরি ঠিকানা রাখে। & দিয়ে ঠিকানা, * দিয়ে সেই ঠিকানার মান।

প্রথম pointer

ঠিকানা আর মান

#include <stdio.h>

int main(void) {
    int x = 42;
    int *p = &x;

    printf("x = %d\n", x);
    printf("address = %p\n", (void *)p);
    printf("*p = %d\n", *p);
    return 0;
}
outputx = 42 address = 0x... *p = 42

&x ঠিকানা, *p সেই ঠিকানার মান।

ঠিকানা লেখা কাগজ

চেনা জিনিসবাড়ির ঠিকানা
→
টেকের নামpointer

কাগজে বাড়ির ঠিকানা লেখা থাকে, বাড়িটা নয়। pointer-এ variable-এর ঠিকানা থাকে, মানটা নয়।

Pointer দিয়ে swap

ফাংশনটা ঠিকানা নেয়

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int *a মানে a হলো int-এর pointer; *a সেই ঘরের মান।

Pointer দিয়ে swap

main থেকে ডাকো

#include <stdio.h>
void swap(int *a, int *b);

int main(void) {
    int x = 3, y = 7;
    swap(&x, &y);
    printf("%d %d\n", x, y);
    return 0;
}
output7 3

আগের স্লাইডের ফাংশনটা এখানে ডাকা হচ্ছে।

অ্যারে আর pointer

একই পরিবার

#include <stdio.h>

int main(void) {
    int a[3] = {10, 20, 30};
    int *p = a;

    printf("%d\n", a[1]);
    printf("%d\n", *(p + 1));
    return 0;
}
output20 20

অ্যারের নাম প্রথম element-এর ঠিকানা।

⚠️

Array decay

ফাংশনে অ্যারে পাঠালে আসলে প্রথম element-এর pointer যায়। তাই length আলাদা পাঠাতে হয়। sizeof(a) ফাংশনের ভেতরে পুরো অ্যারের সাইজ দেয় না।

C-এর স্ট্রিং

শেষে \0

#include <stdio.h>
#include <string.h>

int main(void) {
    char name[] = "Mitu";
    printf("%s\n", name);
    printf("%zu\n", strlen(name));
    return 0;
}
outputMitu 4

স্ট্রিং হলো char-এর অ্যারে, শেষে \0।

নিজে length বের করা

pointer হাঁটা

#include <stdio.h>

int my_len(const char *s) {
    int n = 0;
    while (s[n] != '\0') n++;
    return n;
}

\0 না পাওয়া পর্যন্ত pointer এগোয়।
