State আর render

ছোট অ্যাপের কাঠামো

let expenses = [];

function render() {
  const list = document.querySelector("#list");
  list.innerHTML = "";
  expenses.forEach(e => {
    const li = document.createElement("li");
    li.textContent = `${e.title}: ${e.amount} টাকা`;
    list.appendChild(li);
  });
}

State বদলালে render() ডাকো — পেজ নিজে থেকে বদলায় না।

নতুন খরচ যোগ

ফর্ম submit

form.addEventListener("submit", (event) => {
  event.preventDefault();
  const title = titleInput.value.trim();
  const amount = Number(amountInput.value);

  if (!title || !amount) return;

  expenses.push({ title, amount });
  save();
  render();
  form.reset();
});

আগে যাচাই, তারপর state বদল, তারপর render।

localStorage

পেজ বন্ধ করলেও থাকবে

function save() {
  localStorage.setItem("expenses", JSON.stringify(expenses));
}

function load() {
  const raw = localStorage.getItem("expenses");
  expenses = raw ? JSON.parse(raw) : [];
}

localStorage শুধু স্ট্রিং রাখে, তাই JSON দিয়ে রূপান্তর।

Git = সময়ের সেভ পয়েন্ট

প্রথমে git init, তারপর git add, git commit। প্রতিটি commit একটা ছবি — ভুল হলে পুরনো ছবিতে ফেরা যায়।

প্রথম Git কমান্ড

টার্মিনালে

git init
git add .
git commit -m "first version"
git branch -M main
git remote add origin <repo-url>
git push -u origin main

git add স্টেজ করে, commit সেভ করে, push GitHub-এ পাঠায়।
